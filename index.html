require('dotenv').config(); const express = require('express'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(express.json()); app.post('/api/sync-dns', async (req, res) => { const { domain, records } = req.body; if (!domain || !records) { return res.status(400).json({ error: 'Domain and records are required.' }); } try { const response = await fetch(`https://spaceship.dev/api/v1/dns/records/${domain}`, { method: 'PUT', headers: { 'Content-Type': 'application/json', 'X-API-Key': process.env.SPACESHIP_API_KEY, 'X-API-Secret': process.env.SPACESHIP_API_SECRET }, body: JSON.stringify({ records: records }) }); if (response.ok) { res.status(200).json({ message: 'DNS Records updated successfully!' }); } else { const errorText = await response.text(); res.status(response.status).json({ error: 'Spaceship API Error', details: errorText }); } } catch (error) { console.error('Proxy Error:', error); res.status(500).json({ error: 'Internal Server Error', details: error.message }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`Backend server running on http://localhost:${PORT}`));
